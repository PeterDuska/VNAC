CREATE TABLE daily_outgoing_limit_alerts (
  iban VARCHAR,
  total_outgoing DOUBLE,
  alert_day DATE
) WITH (
  'connector' = 'kafka',
  'topic' = 'daily_outgoing_limit_alerts',
  'properties.bootstrap.servers' = 'broker:9092',
  'properties.group.id' = 'daily-outgoing-limit-reader',
  'scan.startup.mode' = 'earliest-offset',
  'value.format' = 'avro-confluent',
  'value.avro-confluent.url' = 'http://schema-registry:8081'
);


INSERT INTO daily_outgoing_limit_alerts
SELECT
  iban,
  SUM(ABS(amount)) AS total_outgoing,
  CAST(window_start AS DATE) AS alert_day
FROM TABLE(
  TUMBLE(TABLE transactions, DESCRIPTOR(eventTime), INTERVAL '1' DAY)
)
WHERE amount < 0
GROUP BY iban, window_start, window_end
HAVING SUM(ABS(amount)) > 400;
