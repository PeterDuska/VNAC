CREATE TABLE transaction_count_alerts (
  iban VARCHAR,
  window_start TIMESTAMP(3),
  window_end TIMESTAMP(3),
  transaction_count BIGINT,
  PRIMARY KEY (iban, window_start, window_end) NOT ENFORCED
) WITH (
  'connector' = 'upsert-kafka',
  'topic' = 'transaction_count_alerts',
  'properties.bootstrap.servers' = 'broker:9092',
  'properties.group.id' = 'transaction-count-alerts-reader',
  'key.format' = 'json',
  'value.format' = 'avro-confluent',
  'value.avro-confluent.url' = 'http://schema-registry:8081'
);



INSERT INTO transaction_count_alerts
SELECT
  iban,
  window_start,
  window_end,
  COUNT(*) AS transaction_count
FROM TABLE(
  HOP(TABLE transactions, DESCRIPTOR(eventTime), INTERVAL '1' HOUR, INTERVAL '5' HOURS)
)
GROUP BY iban, window_start, window_end
HAVING COUNT(*) > 3;


SELECT * FROM transaction_count_alerts;
